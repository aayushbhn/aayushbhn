// Backend: Attach realtime socket handlers
export function attachRealtime(io: Server) {
  io.on("connection", (socket) => {
    // Join specific event room
    socket.on("join_event", ({ eventId }) => {
      socket.join(`event:${eventId}`);
    });

    // Leave event
    socket.on("leave_event", ({ eventId }) => {
      socket.leave(`event:${eventId}`);
    });
  });
}

// Helper to broadcast updates
export function broadcastEvent(io: Server, eventId: string, eventName: string, payload: any) {
  io.to(`event:${eventId}`).emit(eventName, payload);
}

// Backend endpoint: GET /admin/event/:eventId/overview
export async function fetchEventOverview(eventId: string) {
  const [registrations, tokens, consultations, sales] = await Promise.all([
    prisma.visitorRegistration.count({ where: { eventId } }),
    prisma.tokenBooking.count({ where: { eventId } }),
    prisma.consultationSession.count({ where: { eventId } }),
    prisma.salesAssistance.count({ where: { status: "PURCHASED" } })
  ]);

  return {
    totalRegistrations: registrations,
    totalTokensIssued: tokens,
    consultationsCompleted: consultations,
    salesConversions: sales,
    conversionRate: (consultations / tokens * 100).toFixed(1),
  };
}

// POST /webhooks/orders/create
// Handles order creation from Shopify webhook

async function handleOrderCreate(webhook) {
  const order = webhook.payload;
  const customer = order.customer;
  
  // 1. Check for redemptions (points-based rewards)
  const redeemableVariants = order.line_items
    .filter(item => isRedeemableVariant(item.variant_id));
  
  if (redeemableVariants.length > 0) {
    // Calculate points to deduct based on reward variant
    const pointsToDeduct = calculateRedemptionPoints(redeemableVariants);
    
    // Record deduction in ledger
    await LoyaltyLedger.create({
      customerId: customer.id,
      type: 'REDEEM',
      amount: pointsToDeduct,
      reference: order.id,
    });
  }
  
  // 2. Award purchase points
  const orderTotal = order.total_price;
  const pointsEarned = Math.floor(orderTotal / 10); // $10 = 1 point
  
  await LoyaltyLedger.create({
    customerId: customer.id,
    type: 'EARN',
    amount: pointsEarned,
    reason: 'purchase',
    reference: order.id,
  });
  
  // 3. Sync to Shopify metafield
  await updateCustomerMetafield(customer.id, {
    'nepa_loyalty.points_balance': customer.pointsBalance,
  });
}

  const createDiscountMutation = `
    mutation CreateProductDiscount($input: DiscountCodeBasicInput!) {
      discountCodeBasicCreate(input: $input) {
        discountCodeBasic {
          discountCode {
            code
          }
        }
      }
    }
  `;
  
  // Create discount for 100% off specific variant
  await admin.graphql(createDiscountMutation, {
    variables: {
      input: {
        code: code,
        title: `Rudraksha Reward - ${rewardName}`,
        startsAt: new Date().toISOString(),
        combinesWith: { productsDiscounts: true },
        customerGets: {
          value: { percentage: 1.0 }, // 100% off
          items: { productVariants: { add: [rewardVariantGid] } }
        },
        customerEligibility: { checkoutExclusions: { checkoutOnlineStoreWebUrl: [] } },
        minimumRequirement: { subtotal: { greaterThanOrEqualToSubtotal: 0 } }
      }
    }
  });
  
  return { code, variantGid: rewardVariantGid };
}

    }

        ->groupBy('primary_tag')
        ->get();
    
    return response()->json([
        'intents' => $intents,
        'total_sessions' => $intents->sum('count'),
    ]);
}

class ChatMessageSent implements ShouldBroadcast {
    public function __construct(
        public ChatSession $session,
        public string $message
    ) {}
    
    public function broadcastOn() {
        return new Channel("veda-ai-chat.{$this->session->id}");
    }
    
    public function broadcastAs() {
        return 'chat.message';
    }
